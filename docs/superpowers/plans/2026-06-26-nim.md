```ts
import type { Solution } from '../types/plugin';
import type { NimState, NimMove } from '../types/nim';

export function computeNimSum(piles: number[]): number {
  return piles.reduce((a, b) => a ^ b, 0);
}

export function isPPosition(piles: number[]): boolean {
  return computeNimSum(piles) === 0;
}

export function findWinningMove(piles: number[]): NimMove | null {
  const nimSum = computeNimSum(piles);
  if (nimSum === 0) return null;
  for (let i = 0; i < piles.length; i++) {
    const target = piles[i]! ^ nimSum;
    if (target < piles[i]!) {
      return { pileIndex: i, removeCount: piles[i]! - target };
    }
  }
  return null;
}

export function solveNim(state: NimState): Solution {
  const startTime = performance.now();
  const { piles } = state;
  const nimSum = computeNimSum(piles);
  const winningMove = findWinningMove(piles);
  const pValue = nimSum === 0 ? 0 : 1;
  const xorExpr = piles.join(' ^ ') + ' = ' + nimSum;

  const timeMs = Math.round(performance.now() - startTime);

  return {
    solvable: true,
    moves: winningMove ? [winningMove] : [],
    trace: {
      type: 'nim-xor',
      steps: [{
        description: `Nim-sum: ${xorExpr} → ${pValue === 0 ? 'P-position' : 'N-position'}`,
        data: { nimSum, piles: [...piles], pValue, winningMove },
      }],
      visualizationData: { nimSum, piles: [...piles], pValue, xorExpr },
    },
    stats: {
      nodesExplored: 1,
      timeMs,
      solutionLength: winningMove ? 1 : 0,
      solverName: 'Nim-sum (Bouton 1901)',
      terminated: 'solved',
    },
  };
}
```

```ts
import type { Difficulty } from '../types/plugin';

export function computeNimDifficulty(pileCount: number, maxStones: number): Difficulty {
  const pileFactor = Math.min(1, Math.max(0, (pileCount - 2) / 3));    // 2→0, 5→1
  const stoneFactor = Math.min(1, Math.max(0, (maxStones - 3) / 9));   // 3→0, 12→1
  const score = Math.round((pileFactor * 0.5 + stoneFactor * 0.5) * 100);
  const clamped = Math.max(0, Math.min(100, score));

  let labelKey: string;
  if (clamped <= 33) labelKey = 'difficulty.easy';
  else if (clamped <= 66) labelKey = 'difficulty.medium';
  else labelKey = 'difficulty.hard';

  return {
    score: clamped,
    factors: { pileCount, maxStones },
    labelKey,
  };
}
```

```ts
import type { Level, LevelConfig } from '../types/plugin';
import type { NimState } from '../types/nim';
import { computeNimSum } from '../solver/nim-solver';
import { computeNimDifficulty } from '../difficulty/nim-difficulty';

function mapRangeToPileCount(range: [number, number]): number {
  const mid = (range[0] + range[1]) / 2;
  if (mid <= 33) return 2;
  if (mid <= 66) return 3;
  return 4 + Math.floor(Math.random() * 2); // 4-5
}

function mapRangeToMaxStone(range: [number, number]): number {
  const mid = (range[0] + range[1]) / 2;
  if (mid <= 33) return 3 + Math.floor(Math.random() * 3);   // 3-5
  if (mid <= 66) return 5 + Math.floor(Math.random() * 4);   // 5-8
  return 8 + Math.floor(Math.random() * 5);                   // 8-12
}

function generatePPosition(pileCount: number, maxStone: number): number[] {
  // Generate piles ensuring nim-sum = 0 and all values in [1, maxStone]
  const piles: number[] = [];
  let runningXor = 0;
  for (let i = 0; i < pileCount - 1; i++) {
    const stone = 1 + Math.floor(Math.random() * maxStone);
    piles.push(stone);
    runningXor ^= stone;
  }
  // Last pile: need runningXor ^ x = 0 → x = runningXor, but must be in [1, maxStone]
  if (runningXor >= 1 && runningXor <= maxStone) {
    piles.push(runningXor);
  } else {
    // runningXor is 0 or out of range — adjust a random earlier pile
    const idx = Math.floor(Math.random() * (pileCount - 1));
    const prev = piles[idx]!;
    // newRunningXor = runningXor ^ prev ^ newVal; we want newRunningXor in [1, maxStone]
    const target = 1 + Math.floor(Math.random() * maxStone);
    piles[idx] = target;
    const newRunningXor = runningXor ^ prev ^ target;
    piles.push(Math.max(1, Math.min(maxStone, newRunningXor)));
  }
  return piles;
}

export function generateNimLevel(config: LevelConfig): Level {
  const pileCount = mapRangeToPileCount(config.difficultyRange);
  const maxStone = mapRangeToMaxStone(config.difficultyRange);

  // Generate P-position then walk 1 step to N-position
  const pPiles = generatePPosition(pileCount, maxStone);
  // Walk 1 step: pick a pile with stones, remove some (not all, unless last)
  const piles = [...pPiles];
  const nonEmptyIndices = piles.map((v, i) => v > 0 ? i : -1).filter(i => i >= 0);
  const walkPile = nonEmptyIndices[Math.floor(Math.random() * nonEmptyIndices.length)]!;
  const removeCount = 1 + Math.floor(Math.random() * (piles[walkPile]! - 1)) || 1;
  piles[walkPile]! -= removeCount;

  const difficulty = computeNimDifficulty(pileCount, maxStone);

  return {
    schemaVersion: 1,
    gameId: 'nim',
    levelId: `nim-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    initialState: {
      piles,
      currentPlayer: 0,
      initialPiles: [...piles],
    } as unknown as Record<string, unknown>,
    difficulty,
    optimalSteps: null,
    uniqueSolution: false,
    generatedBy: 'p-distance-walk',
    createdAt: new Date().toISOString(),
  };
}
```

```ts
import type { Difficulty, Solution } from '@ludus/core';
import type { NimState, NimMove } from '@ludus/core';
import { computeNimSum, solveNim as coreSolveNim } from '@ludus/core';
import { computeNimDifficulty } from '@ludus/core';

export type { NimState, NimMove };

export function createInitialState(): NimState {
  return {
    piles: [3, 4, 5],
    currentPlayer: 0,
    initialPiles: [3, 4, 5],
  };
}

export function legalMoves(state: NimState): NimMove[] {
  if (state.currentPlayer !== 0) return [];
  const moves: NimMove[] = [];
  for (let i = 0; i < state.piles.length; i++) {
    const count = state.piles[i]!;
    for (let r = 1; r <= count; r++) {
      moves.push({ pileIndex: i, removeCount: r });
    }
  }
  return moves;
}

export function applyMove(state: NimState, move: NimMove): NimState {
  const { piles, currentPlayer } = state;
  const idx = move.pileIndex;
  if (idx < 0 || idx >= piles.length) return state;
  if (move.removeCount < 1 || move.removeCount > piles[idx]!) return state;
  // In human vs AI mode, player is always 0 when calling applyMove
  if (currentPlayer !== 0) return state;

  const newPiles = [...piles];
  newPiles[idx]! -= move.removeCount;

  return {
    piles: newPiles,
    currentPlayer: 1, // switch to AI
    initialPiles: state.initialPiles,
  };
}

export function isSolved(state: NimState): boolean {
  // Player wins: all piles empty and it's AI's turn (player took last stone)
  return state.piles.every(v => v === 0) && state.currentPlayer === 1;
}

export function getDifficulty(state: NimState): Difficulty {
  const pileCount = state.piles.length;
  const maxStones = Math.max(...state.piles);
  return computeNimDifficulty(pileCount, maxStones);
}

export function solveNimState(state: NimState): Solution {
  // Solve uses the current piles regardless of currentPlayer
  const tempState: NimState = { ...state, currentPlayer: 0 };
  return coreSolveNim(tempState);
}
```

```markdown
# Nim 取石子 — XOR 的博弈魔法

## 1. 玩

Nim 是世界上最简单的博弈游戏之一：几堆石子，两人轮流从某一堆中取走任意数量（至少 1 颗，可以取完整堆）。取走最后一颗石子的人获胜。

试试看！你能找到战胜 AI 的秘诀吗？

## 2. 思考

- 为什么 [1, 2, 3] 这样的局面，无论你怎么走，都会输？
- 为什么 [3, 4, 5] 你却有机会赢？
- 有没有一种"万能公式"，看一眼就知道胜负？

## 3. 揭示

1901 年，哈佛数学家 Charles Bouton 发现了 Nim 的必胜秘诀——**Nim-sum**：

$$\text{Nim-sum} = a_1 \oplus a_2 \oplus \cdots \oplus a_k$$

其中 $\oplus$ 是 XOR（按位异或）运算。

- **Nim-sum = 0 → P 位置（Previous player win）**：上一个走的人必胜，轮到你时已经晚了
- **Nim-sum ≠ 0 → N 位置（Next player win）**：你必胜！只要找一堆，使取走后剩余的数等于 `当前堆数 XOR nim-sum`

**直觉**：XOR = 0 就像一个"完美平衡" —— 无论你怎么打破，对手都能一键恢复平衡。

## 4. 求解

对于局面 [3, 4, 5]：

```
3 = 011 (二进制)
4 = 100
5 = 101
XOR = 010 = 2 ≠ 0 → N 位置，你必胜！

必胜走法：找堆 3 (011)
011 ⊕ 010 = 001 = 1 < 3
从堆 3 取 2 颗 → 局面变为 [1, 4, 5]
验证：1 ⊕ 4 ⊕ 5 = 000 = 0 → 对手陷入 P 位置！
```

## 5. 出题

Nim 的关卡生成简单而优雅：先随机生成一个 P 位置（Nim-sum = 0），然后随机走一步得到 N 位置作为考题。难度由堆数和石子数控制——堆越多、数字越大，XOR 心算越难。
```

```tsx
import dagre from 'dagre';
import type { SolverTrace } from '@ludus/core';
import { computeNimSum, findWinningMove } from '@ludus/core';
import type { NimState, NimMove } from '@ludus/games';

interface Props {
  trace: SolverTrace;
  state: NimState;
}

interface TreeNodeData {
  piles: number[];
  nimSum: number;
  isP: boolean;
  move?: string;
}

function expandTree(piles: number[], depth: number, maxDepth: number): { nodes: TreeNodeData[]; edges: { from: number; to: number; label: string }[] } {
  const nodes: TreeNodeData[] = [];
  const edges: { from: number; to: number; label: string }[] = [];
  const queue: { piles: number[]; parentIdx: number | null; depth: number; moveLabel: string }[] = [
    { piles, parentIdx: null, depth: 0, moveLabel: '' }
  ];

  while (queue.length > 0) {
    const item = queue.shift()!;
    if (item.depth > maxDepth) continue;

    const nimSum = computeNimSum(item.piles);
    const nodeIdx = nodes.length;
    nodes.push({ piles: [...item.piles], nimSum, isP: nimSum === 0, move: item.moveLabel || undefined });

    if (item.parentIdx !== null) {
      edges.push({ from: item.parentIdx, to: nodeIdx, label: item.moveLabel });
    }

    if (item.depth < maxDepth) {
      for (let i = 0; i < item.piles.length; i++) {
        for (let r = 1; r <= item.piles[i]!; r++) {
          const newPiles = [...item.piles];
          newPiles[i]! -= r;
          queue.push({ piles: newPiles, parentIdx: nodeIdx, depth: item.depth + 1, moveLabel: `${i + 1}:${r}` });
        }
      }
    }
  }

  return { nodes, edges };
}

function layoutDagre(nodes: TreeNodeData[], edges: { from: number; to: number; label: string }[]) {
  const g = new dagre.graphlib.Graph();
  g.setDefaultEdgeLabel(() => ({}));
  g.setGraph({ rankdir: 'TB', nodesep: 20, ranksep: 60 });

  nodes.forEach((_, i) => {
    const w = 80;
    const h = 40;
    g.setNode(String(i), { width: w, height: h });
  });

  edges.forEach(e => {
    g.setEdge(String(e.from), String(e.to));
  });

  dagre.layout(g);

  const rfNodes: Node[] = nodes.map((n, i) => {
    const pos = g.node(String(i));
    return {
      id: String(i),
      position: { x: pos.x - 40, y: pos.y - 20 },
      data: { label: `${n.piles.join(',')}\n${n.nimSum} ${n.isP ? 'P' : 'N'}` },
      style: {
        background: n.isP ? '#fef2f2' : '#eff6ff',
        border: `2px solid ${n.isP ? '#e05d5d' : '#4a90d9'}`,
        borderRadius: '8px',
        padding: '6px 10px',
        fontSize: '11px',
        fontFamily: 'JetBrains Mono, monospace',
        textAlign: 'center',
      },
      sourcePosition: Position.Bottom,
      targetPosition: Position.Top,
    };
  });

  const rfEdges: Edge[] = edges.map((e, i) => ({
    id: `e${i}`,
    source: String(e.from),
    target: String(e.to),
    label: e.label,
    style: { stroke: '#a8a29e', strokeWidth: 1 },
    labelStyle: { fontSize: 9, fill: '#78716c' },
    labelBgStyle: { fill: '#fafaf9' },
  }));

  return { nodes: rfNodes, edges: rfEdges };
}

export function NimTreeViz({ trace, state }: Props) {
  const { t } = useTranslation();

  const tree = useMemo(() => {
    const { nodes, edges } = expandTree(state.piles, 0, 3);
    return layoutDagre(nodes, edges);
  }, [state.piles]);

  const [nodes, setNodes, onNodesChange] = useNodesState(tree.nodes);
  const [edges, setEdges, onEdgesChange] = useEdgesState(tree.edges);

  return (
    <div className="bento-card p-4 space-y-3">
      <h3 className="text-sm font-semibold text-warm-stone-700">{t('nim.gameTree')}</h3>
      <div className="text-xs text-warm-stone-500 space-y-1">
        <p>Nim-sum: {computeNimSum(state.piles)} ({computeNimSum(state.piles) === 0 ? t('nim.pPosition') : t('nim.nPosition')})</p>
        <p>{t('nim.nimSum')} = {state.piles.join(' ^ ')} = {computeNimSum(state.piles)}</p>
      </div>
      <div style={{ width: '100%', height: 400 }}>
        <ReactFlow
          nodes={nodes}
          edges={edges}
          onNodesChange={onNodesChange}
          onEdgesChange={onEdgesChange}
          fitView
          nodesDraggable={false}
          zoomOnScroll={false}
          proOptions={{ hideAttribution: true }}
        />
      </div>
      <div className="flex gap-4 text-xs">
        <span className="flex items-center gap-1">
```
